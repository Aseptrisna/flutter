Mood Tracker: CRUD Catatan Mood

Menambahkan tabel milik project Mood Tracker di atas fondasi modul user, memakai Express, Prisma, dan Swagger.

01Yang Akan Dibuat

CRUD tabel catatan_mood untuk project Mood Tracker. Setiap data punya pemilik: pengguna hanya bisa melihat dan mengubah datanya sendiri.

EndpointFungsi

GET /api/catatan-mood/{id}Detail satu data
POST /api/catatan-moodTambah

03Skema & Migrasi

Ganti seluruh isi prisma/schema.prisma. Yang baru: model CatatanMood dan satu baris relasi catatanMood CatatanMood[] di model User.

  catatanMood CatatanMood[]

model CatatanMood {
  id         Int      @id @default(autoincrement())
  user_id    Int
  tanggal    DateTime @db.Date
  mood       String   @db.VarChar(30)
  catatan    String?
  created_at DateTime @default(now()) @db.Timestamp(6)

  user User @relation(fields: [user_id], references: [id], onDelete: Cascade)

  @@index([user_id])
  @@map("catatan_mood")
}

04Repository — Akses Database

Hanya lapisan ini yang berbicara dengan database. Semua daftar dibatasi user_id.

src/repositories/catatanMoodRepository.js
const prisma = require('../db');

// Kolom yang dikirim ke klien. created_at sengaja tidak ikut.
const kolom = {
  id: true,
  user_id: true,
  tanggal: true,
  mood: true,
  catatan: true,
};

// mapRow mengubah tipe dari Prisma menjadi bentuk JSON yang ramah klien:
// Decimal -> angka biasa, tanggal -> teks YYYY-MM-DD.
function mapRow(row) {
  if (!row) return row;
  return {
    ...row,
    tanggal: row.tanggal && row.tanggal.toISOString().slice(0, 10),
  };
}

// Selalu dibatasi user_id, supaya pengguna hanya melihat datanya sendiri.
function buildWhere(userId, q) {
  const where = { user_id: userId };
  if (q) where.catatan = { contains: q, mode: 'insensitive' };
  return where;
}

async function create(data) {
  return mapRow(await prisma.catatanMood.create({ data, select: kolom }));
}

async function findAllByUser(userId, q, limit, offset) {
  const rows = await prisma.catatanMood.findMany({
    where: buildWhere(userId, q),
    orderBy: { id: 'desc' },
    take: limit,
    skip: offset,
    select: kolom,
  });
  return rows.map(mapRow);
}

async function countByUser(userId, q) {
  return prisma.catatanMood.count({ where: buildWhere(userId, q) });
}

async function findById(id) {
  return mapRow(await prisma.catatanMood.findUnique({ where: { id }, select: kolom }));
}

async function update(id, data) {
  return mapRow(await prisma.catatanMood.update({ where: { id }, data, select: kolom }));
}

async function remove(id) {
  await prisma.catatanMood.delete({ where: { id } });
}

module.exports = { create, findAllByUser, countByUser, findById, update, remove };

05Service — Validasi & Hak Akses

Objek aturan menentukan isian yang sah; validate membuang kolom lain yang dikirim klien. authorizeOwner membalas 404 bila data tidak ada dan 403 bila milik orang lain. PUT menulis ulang seluruh data, jadi kolom opsional yang tidak dikirim menjadi kosong (atau false).

src/services/catatanMoodService.js
const catatanMoodRepository = require('../repositories/catatanMoodRepository');
const { normalizePagination } = require('../utils/pagination');
const { validate } = require('../utils/validate');
const { NotFoundError, ForbiddenError } = require('../errors');

// Aturan tiap kolom. Samakan dengan schema.prisma.
const aturan = {
  tanggal: { tipe: 'tanggal', wajib: true },
  mood: { tipe: 'pilihan', wajib: true, opsi: ['senang', 'biasa', 'sedih', 'marah', 'cemas'] },
  catatan: { tipe: 'teks', wajib: false },
};

async function list(userId, q, rawPage, rawLimit) {
  const { page, limit } = normalizePagination(rawPage, rawLimit);
  const offset = (page - 1) * limit;

  const data = await catatanMoodRepository.findAllByUser(userId, q, limit, offset);
  const total = await catatanMoodRepository.countByUser(userId, q);
  return { data, total, page, limit };
}

// authorizeOwner: datanya harus ada (404) dan milik pengguna yang login (403).
async function authorizeOwner(userId, id) {
  const data = await catatanMoodRepository.findById(id);
  if (!data) throw new NotFoundError('data tidak ditemukan');
  if (data.user_id !== userId) throw new ForbiddenError('Anda bukan pemilik data ini');
  return data;
}

async function create(userId, body) {
  const data = validate(body, aturan);
  // user_id diambil dari token, bukan dari isian klien
  return catatanMoodRepository.create({ ...data, user_id: userId });
}

async function update(userId, id, body) {
  await authorizeOwner(userId, id);
  return catatanMoodRepository.update(id, validate(body, aturan));
}

async function remove(userId, id) {
  await authorizeOwner(userId, id);
  await catatanMoodRepository.remove(id);
}

module.exports = { list, detail: authorizeOwner, create, update, remove };

06Handler — Request & Response

Mengambil input dari req, memanggil service, lalu membalas JSON.

src/handlers/catatanMoodHandler.js
const catatanMoodService = require('../services/catatanMoodService');
const { ok, paginated } = require('../utils/response');
const { parseId } = require('../utils/validate');

async function list(req, res, next) {
  try {
    const q = typeof req.query.q === 'string' ? req.query.q : '';
    const { data, total, page, limit } = await catatanMoodService.list(req.userId, q, req.query.page, req.query.limit);
    paginated(res, 200, data, page, limit, total);
  } catch (err) {
    next(err);
  }
}

async function detail(req, res, next) {
  try {
    ok(res, 200, await catatanMoodService.detail(req.userId, parseId(req.params.id)));
  } catch (err) {
    next(err);
  }
}

async function create(req, res, next) {
  try {
    ok(res, 201, await catatanMoodService.create(req.userId, req.body));
  } catch (err) {
    next(err);
  }
}

async function update(req, res, next) {
  try {
    ok(res, 200, await catatanMoodService.update(req.userId, parseId(req.params.id), req.body));
  } catch (err) {
    next(err);
  }
}

async function remove(req, res, next) {
  try {
    await catatanMoodService.remove(req.userId, parseId(req.params.id));
    ok(res, 200, { message: 'data berhasil dihapus' });
  } catch (err) {
    next(err);
  }
}

module.exports = { list, detail, create, update, remove };

07Route

Ganti seluruh isi src/app.js. Tambahannya: satu require handler dan lima route yang semuanya melewati requireAuth.

src/app.js
const express = require('express');
const cors = require('cors');
const swaggerUi = require('swagger-ui-express');

const swaggerSpec = require('./docs/swagger');
const { requireAuth } = require('./middleware/auth');
const { loginLimiter } = require('./middleware/rateLimit');
const { errorHandler } = require('./middleware/errorHandler');

const authHandler = require('./handlers/authHandler');
const catatanMoodHandler = require('./handlers/catatanMoodHandler');

const app = express();

app.use(cors());
app.use(express.json());

// auth
app.post('/api/auth/register', authHandler.register);
app.post('/api/auth/login', loginLimiter, authHandler.login);

// profile (butuh login)
app.get('/api/profile', requireAuth, authHandler.profile);
app.put('/api/profile', requireAuth, authHandler.updateProfile);
app.put('/api/profile/password', requireAuth, authHandler.updatePassword);

// --- route project-mu ---
app.get('/api/catatan-mood', requireAuth, catatanMoodHandler.list);
app.get('/api/catatan-mood/:id', requireAuth, catatanMoodHandler.detail);
app.post('/api/catatan-mood', requireAuth, catatanMoodHandler.create);
app.put('/api/catatan-mood/:id', requireAuth, catatanMoodHandler.update);
app.delete('/api/catatan-mood/:id', requireAuth, catatanMoodHandler.remove);

// health check
app.get('/healthz', (req, res) => res.sendStatus(200));

// dokumentasi Swagger. persistAuthorization: token tetap tersimpan setelah reload
// /swagger/index.html (alamat di versi Go) dialihkan, karena file bawaan paket
// di alamat itu menampilkan contoh "Petstore", bukan API kita.
app.get('/swagger/index.html', (req, res) => res.redirect('/swagger/'));
app.use(
  '/swagger',
  swaggerUi.serve,
  swaggerUi.setup(swaggerSpec, { swaggerOptions: { persistAuthorization: true } }),
);

// errorHandler WAJIB didaftarkan PALING TERAKHIR -- Express mengenali
// middleware 4-parameter (err, req, res, next) sebagai error handler hanya
// kalau ia didaftarkan setelah semua route lain.
app.use(errorHandler);

module.exports = app;
